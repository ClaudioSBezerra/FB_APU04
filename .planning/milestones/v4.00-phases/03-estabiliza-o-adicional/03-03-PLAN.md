FUNÇÕES EXPORTADAS DE formatFilial.ts (11 funções):
  formatCPF(cpf: string | null | undefined): string
    — 11 dígitos → "000.000.000-00", null/undefined → "", não-11-dígitos → valor original
  formatCNPJ(cnpj: string | null | undefined): string
    — 14 dígitos → "00.000.000/0000-00", null/undefined → "", não-14-dígitos → valor original
  formatDocumento(doc: string | null | undefined): string
    — delega para formatCPF ou formatCNPJ baseado em len(doc)
  formatCPFMasked(cpf: string | null | undefined): string
    — 11 dígitos → "***.***.000-**" (exibe apenas 3 dígitos do meio)
  formatCNPJMasked(cnpj: string | null | undefined): string
    — 14 dígitos → "**.***.***/0000-**" (exibe apenas raiz)
  formatDocumentoMasked(doc: string | null | undefined): string
    — delega para masked conforme tamanho
  formatFilialDisplay(codEst: string|null|undefined, cnpj: string|null|undefined): string
    — concatena codEst + " - " + CNPJ formatado; fallback se ambos nulos
  formatFilialDisplayFormatted(codEst, cnpj): string
    — similar, com formatação adicional
  parseFilialName(filialNome: string): { codEst: string | null; cnpj: string | null }
    — parse de strings como "FC010102 - 12345678000190" → {codEst:"FC010102", cnpj:"12345678000190"}
  formatFilialFromRow(row: {...}): string
    — lógica de fallback complexa; testar os caminhos: com codEst, sem codEst com cnpj, sem ambos
  formatCnpjComApelido(cnpj, apelido): string
    — combina CNPJ formatado com apelido da filial
