  function fmtBRL(v: number | null | undefined): string {
    if (v == null) return '—';
    return v.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' });
  }
  function fmtCNPJ(v: string): string {
    if (!v || v.length !== 14) return v || '—';
    return `${v.slice(0, 2)}.${v.slice(2, 5)}.${v.slice(5, 8)}/${v.slice(8, 12)}-${v.slice(12)}`;
  }
  function buildUrl(base: string, params: Record<string, string>): string {
    const q = new URLSearchParams(Object.entries(params).filter(([, v]) => v !== ''));
    return q.toString() ? `${base}?${q}` : base;
  }

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-02-01 | Information Disclosure | fetch sem auth token | accept | window.fetch interceptado por AuthContext (decisão STATE.md) — Authorization + X-Company-ID injetados automaticamente em todo fetch global |
| T-04-02-02 | Tampering | mes_ano digitado pelo usuário | mitigate | mes_ano passado como query param ao backend; backend valida e parametriza via $N — frontend não executa SQL |
| T-04-02-03 | Information Disclosure | CSV download — arquivo fiscal local | accept | Download ocorre no contexto do usuário autenticado; o arquivo fica na máquina do auditor, que é o caso de uso intencional para auditoria |
| T-04-02-04 | Denial of Service | exportToExcel com divergencias muito grandes | mitigate | Backend retorna LIMIT 500 — máximo 500 rows; xlsx client-side processa esse volume sem problema |
| T-04-02-05 | Information Disclosure | window.print() expõe dados na impressora | accept | Funcionalidade de impressão é requisito explícito do auditor (EXP-01); dados fiscais da própria empresa |
</threat_model>
